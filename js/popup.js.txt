/**
 * Fungsi utama yang dipanggil saat area kabupaten diklik
 * @param {Object} e - Event klik Leaflet
 * @param {Object} feature - Fitur GeoJSON administrasi yang diklik
 */
function handleMapClick(e, feature) {
    let lat = e.latlng.lat;
    let lon = e.latlng.lng;
    
    let props = feature.properties || {};
    
    // Identifikasi Wilayah (Gunakan Kode BPS/Kemendagri atau Nama Kabupaten)
    let kodeWilayah = props.KODBPS || props.KODENC || props.KAB_CODE || props.id || '';
    let namaWilayah = props.WADMKK || props.kabupaten || props.NAME_2 || "Wilayah Tidak Diketahui";
    let namaProvinsi = props.WADMPR || props.provinsi || props.NAME_1 || "";
    
    let coordText = (typeof Utils !== 'undefined' && typeof Utils.formatKoordinat === 'function')
        ? `${Utils.formatKoordinat(lat, 'lat')}, ${Utils.formatKoordinat(lon, 'lon')}`
        : `${lat.toFixed(4)}, ${lon.toFixed(4)}`;

    let productConfig = CONFIG.products[currentCategory][currentProductKey];
    let productTitle = productConfig.title || productConfig.name || "Analisis Risiko";
    
    // Generasi tabel multi-hari (H-1 s.d H+5) berdasarkan data olahan backend
    let tableHTML = generateImpactTableFromBackendData(kodeWilayah, namaWilayah, productConfig);

    let popupContent = `
        <div class="impact-popup">
            <div class="popup-header">
                <div class="popup-title">📍 ${namaWilayah.toUpperCase()}</div>
                <div class="popup-subtitle">${namaProvinsi}</div>
                <div class="popup-coords">${coordText}</div>
            </div>
            
            <div class="popup-body">
                <div class="product-info">
                    <span class="info-label">Analisis:</span> 
                    <span class="info-value">${productTitle}</span>
                </div>
                
                <div class="table-container">
                    <table class="impact-table">
                        <thead>
                            <tr>
                                <th>Hari</th>
                                <th>Tanggal Valid</th>
                                <th>Tingkat Status</th>
                            </tr>
                        </thead>
                        <tbody>
                            ${tableHTML}
                        </tbody>
                    </table>
                </div>
            </div>
            
            <div class="popup-footer">
                <button class="btn-export" onclick="exportImpactReportPDF('${namaWilayah}', '${kodeWilayah}')">
                    📄 Unduh Laporan (PDF)
                </button>
            </div>
        </div>
    `;

    L.popup({
        maxWidth: 380,
        minWidth: 320,
        className: 'futuristic-popup'
    })
    .setLatLng(e.latlng)
    .setContent(popupContent)
    .openOn(map);
}

/**
 * Membaca data status per hari berdasarkan ID/Nama Wilayah dari Rekap Backend
 */
function generateImpactTableFromBackendData(kodeWilayah, namaWilayah, productConfig) {
    if (!window.validDates || window.validDates.length === 0) {
        return `<tr><td colspan="3" style="text-align:center; padding: 10px;">Data tanggal tidak tersedia.</td></tr>`;
    }

    let html = '';

    window.validDates.forEach((dateStr, index) => {
        let dayLabel = index === 0 ? 'H-1' : `H+${index - 1}`;
        let formattedDate = (typeof Utils !== 'undefined' && typeof Utils.formatTanggal === 'function') 
            ? Utils.formatTanggal(dateStr) 
            : dateStr;
        
        let statusValue = "Aman / Normal";
        
        // Pencarian ekstra luwes dari window.backendImpactData
        if (window.backendImpactData) {
            let regionData = window.backendImpactData[kodeWilayah] 
                          || window.backendImpactData[namaWilayah]
                          || window.backendImpactData[namaWilayah.toUpperCase()]
                          || window.backendImpactData[namaWilayah.toLowerCase()];

            if (regionData && regionData[`day_${index}`]) {
                statusValue = regionData[`day_${index}`];
            }
        }

        let badgeStyle = getBadgeStyle(statusValue, productConfig);

        html += `
            <tr>
                <td class="day-label">${dayLabel}</td>
                <td class="date-label">${formattedDate}</td>
                <td class="status-cell">
                    <span class="badge" style="background: ${badgeStyle.bg}; color: ${badgeStyle.color}; border: 1px solid ${badgeStyle.border};">
                        ${statusValue}
                    </span>
                </td>
            </tr>
        `;
    });

    return html;
}

/**
 * Menyesuaikan warna badge sesuai level di config
 */
function getBadgeStyle(statusText, productConfig) {
    let text = String(statusText).toLowerCase();
    let matchedLegend = null;

    if (productConfig && Array.isArray(productConfig.legends)) {
        matchedLegend = productConfig.legends.find(l => 
            (l.level && text.includes(String(l.level).toLowerCase())) || 
            (l.label && text.includes(String(l.label).toLowerCase()))
        );
    }
    
    if (matchedLegend && matchedLegend.color && matchedLegend.color !== 'transparent') {
        return {
            bg: hexToRgba(matchedLegend.color, 0.2),
            color: matchedLegend.color,
            border: matchedLegend.color
        };
    }

    // Default styling jika status normal/aman
    return {
        bg: 'rgba(34, 197, 94, 0.15)',
        color: '#4ade80',
        border: '#22c55e'
    };
}

function hexToRgba(hex, alpha) {
    if (!hex || hex === 'transparent') return 'transparent';
    let c = hex.replace('#','');
    if (c.length === 3) c = c.split('').map(x => x + x).join('');
    let num = parseInt(c, 16);
    return `rgba(${(num >> 16)&255}, ${(num >> 8)&255}, ${num&255}, ${alpha})`;
}
